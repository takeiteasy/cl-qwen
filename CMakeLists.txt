cmake_minimum_required(VERSION 3.16)
project(cl_qwen C)
set(CMAKE_C_STANDARD 99)
add_library(cl_qwen_numeric SHARED native/q8.c)
target_compile_options(cl_qwen_numeric PRIVATE -O3 -Wall -Wextra -Werror -ffp-contract=off)
if(NOT APPLE)
  target_link_libraries(cl_qwen_numeric PRIVATE m)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(q8_test native/test_q8.c native/q8.c)
  add_executable(q8_scalar_test native/test_q8.c native/q8.c)
  target_compile_definitions(q8_scalar_test PRIVATE CQ_SCALAR=1)
  foreach(target q8_test q8_scalar_test)
    target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -Werror -ffp-contract=off)
    if(NOT APPLE)
      target_link_libraries(${target} PRIVATE m)
    endif()
    add_test(NAME ${target} COMMAND ${target})
  endforeach()
endif()
